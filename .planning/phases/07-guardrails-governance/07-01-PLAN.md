**SHARED HELPER FUNCTION** — define once in each wrapper, identical across all three. This is the core of the check:

```bash
# ── GOVERNANCE PRE-FLIGHT ───────────────────────────────────────────────
# Phase 7: Enforce wrapper_allowlist from the active governance YAML file.
# Silent pass-through if HERMES_LAB_GOVERNANCE is not set (backward compat).
GOVERNANCE="${HERMES_LAB_GOVERNANCE:-}"
if [[ -n "$GOVERNANCE" ]]; then
  TRACK="${HERMES_LAB_TRACK:-}"
  COURSE_DIR="$(cd "$(dirname "$0")/../.." && pwd)"

  # Resolve governance file path
  if [[ "$GOVERNANCE" == "L4" ]]; then
    if [[ -z "$TRACK" ]]; then
      printf 'WARNING: HERMES_LAB_GOVERNANCE=L4 requires HERMES_LAB_TRACK=track-{a,b,c}. Defaulting to track-a.\n' >&2
      TRACK="track-a"
    fi
    GOVERNANCE_FILE="${COURSE_DIR}/governance/governance-L4-${TRACK}.yaml"
  else
    GOVERNANCE_FILE="${COURSE_DIR}/governance/governance-${GOVERNANCE}.yaml"
  fi

  if [[ ! -f "$GOVERNANCE_FILE" ]]; then
    printf 'GOVERNANCE ERROR: Cannot find %s\n' "$GOVERNANCE_FILE" >&2
    exit 1
  fi

  # Extract wrapper_allowlist for THIS tool from the yaml file using awk.
  # Tool key is set per-wrapper: TOOL_KEY=kubectl|aws|psql
  ALLOWED_PREFIXES=$(awk -v tool="  ${TOOL_KEY}:" '
    /^wrapper_allowlist:/ { in_wa=1; next }
    in_wa && $0 ~ "^"tool"$" { in_tool=1; next }
    in_tool && /^    - / { gsub(/^    - "|"$/, ""); print; next }
    in_tool && /^  [a-z]/ { in_tool=0 }
    in_wa && /^[a-z]/ { in_wa=0; in_tool=0 }
  ' "$GOVERNANCE_FILE")

  # Check if the current command prefix matches any allowed prefix
  # CMD_CHECK is set per-wrapper (different extraction per tool — see per-wrapper sections)
  ALLOWED=0
  while IFS= read -r prefix; do
    [[ -z "$prefix" ]] && continue
    if [[ "$CMD_CHECK" == "$prefix"* ]]; then
      ALLOWED=1
      break
    fi
  done <<< "$ALLOWED_PREFIXES"

  if [[ "$ALLOWED" == "0" ]]; then
    printf '\n' >&2
    printf '╔══════════════════════════════════════════════════╗\n' >&2
    printf '║           [ GOVERNANCE REJECTED ]                ║\n' >&2
    printf '╠══════════════════════════════════════════════════╣\n' >&2
    printf '║ Command:      %s %s\n' "${TOOL_KEY}" "$CMD_CHECK" >&2
    printf '║ Governance:   %s' "$GOVERNANCE" >&2
    [[ -n "$TRACK" ]] && printf ' (%s)' "$TRACK" >&2
    printf '\n' >&2
    printf '║ Block reason: Not in %s wrapper_allowlist.%s\n' "$GOVERNANCE" "$TOOL_KEY" >&2
    printf '║ Governance file: %s\n' "$GOVERNANCE_FILE" >&2
    printf '║ To unblock:   Upgrade HERMES_LAB_GOVERNANCE or add to wrapper_allowlist.\n' >&2
    printf '║ SOUL.md:      Behavioral NEVER rules may also apply at Layer 3.\n' >&2
    printf '╚══════════════════════════════════════════════════╝\n' >&2
    printf '\n' >&2
    exit 1
  fi
fi
# ────────────────────────────────────────────────────────────────────────
```
