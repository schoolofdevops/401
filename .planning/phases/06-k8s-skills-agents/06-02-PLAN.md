**File 3a: `infrastructure/scenarios/k8s/03-oom-killed.yaml`**

Use the Python Alpine bytearray pattern from RESEARCH.md Pitfall 4 (more reliable on Apple Silicon arm64 than busybox dd):

```yaml
# Scenario 03: OOMKilled
# Failure mode: Container allocates 64MB heap against a 32Mi memory limit -> cgroup OOM kill -> exitCode 137
# Apple Silicon note: We use python:3.12-alpine bytearray instead of busybox dd because dd is I/O-bound
# and may not trigger cgroup OOM on arm64/macOS Docker (RESEARCH Pitfall 4).
# Apply: kubectl apply -f infrastructure/scenarios/k8s/03-oom-killed.yaml
# Cleanup: kubectl delete namespace k8s-trouble-oom
# YAML_DOC_SEPARATOR — when authoring the file, replace this comment with three literal hyphens (---) on its own line
apiVersion: v1
kind: Namespace
metadata:
  name: k8s-trouble-oom
  labels:
    scenario: k8s-trouble
    failure-mode: oom-killed
# YAML_DOC_SEPARATOR — when authoring the file, replace this comment with three literal hyphens (---) on its own line
apiVersion: apps/v1
kind: Deployment
metadata:
  name: memory-eater
  namespace: k8s-trouble-oom
spec:
  replicas: 1
  selector:
    matchLabels:
      app: memory-eater
  template:
    metadata:
      labels:
        app: memory-eater
    spec:
      containers:
      - name: memory-eater
        image: python:3.12-alpine
        command: ["python3", "-c"]
        args:
        - |
          import time
          # Allocate 64 MB of heap — exceeds the 32Mi cgroup limit -> OOMKilled with exitCode 137
          b = bytearray(64 * 1024 * 1024)
          time.sleep(3600)
        resources:
          requests:
            memory: "16Mi"
          limits:
            memory: "32Mi"
```

**Step 2: Create `infrastructure/scenarios/k8s/capture-mock-data.sh`**

This script automates the live-capture workflow described in CONTEXT.md D-13. Even if mock files end up hand-authored in this task, the script is the canonical source for re-capturing them from a real KIND cluster.

```bash
#!/usr/bin/env bash
# capture-mock-data.sh — Capture real kubectl outputs from KIND into mock JSON parity files
#
# Usage:
#   1. Ensure KIND cluster is running: kind get clusters
#   2. Apply each scenario manifest, wait for it to reach the failure state, capture kubectl outputs
#   3. Repeat for all 6 scenarios
#
# Each scenario:
#   - Creates a dedicated namespace (k8s-trouble-*)
#   - Captures: get pods -o json, describe pod (text), and where applicable: logs --previous, get endpoints
#   - Writes to infrastructure/mock-data/kubernetes/<NN>-<scenario>-<command>.json|txt
#
# After successful capture: this script does NOT clean up — run `kubectl delete namespace k8s-trouble-*`
# manually so you can inspect the failures yourself.

set -euo pipefail

REPO_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/../../.." && pwd)"
SCENARIOS_DIR="${REPO_ROOT}/infrastructure/scenarios/k8s"
MOCK_DIR="${REPO_ROOT}/infrastructure/mock-data/kubernetes"

mkdir -p "$MOCK_DIR"

if ! command -v kubectl &> /dev/null; then
  echo "ERROR: kubectl not found in PATH" >&2
  exit 1
fi

if ! kubectl cluster-info &> /dev/null; then
  echo "ERROR: kubectl cannot reach a cluster. Start KIND first: kind create cluster --config infrastructure/kind/cluster-config.yaml" >&2
  exit 1
fi

wait_for_state() {
  local ns="$1"
  local expected_status="$2"
  local timeout="${3:-120}"
  local elapsed=0
  echo "Waiting for $ns to reach $expected_status (timeout ${timeout}s)..."
  while [ $elapsed -lt $timeout ]; do
    if kubectl get pods -n "$ns" 2>/dev/null | grep -q "$expected_status"; then
      echo "Reached $expected_status after ${elapsed}s"
      return 0
    fi
    sleep 5
    elapsed=$((elapsed + 5))
  done
  echo "WARNING: Timed out waiting for $expected_status in $ns" >&2
  return 1
}

capture_scenario() {
  local manifest="$1"
  local ns="$2"
  local prefix="$3"
  local expected_status="$4"
  shift 4
  local extra_commands=("$@")

  echo "=== Capturing scenario: $prefix ==="
  kubectl apply -f "${SCENARIOS_DIR}/${manifest}"
  wait_for_state "$ns" "$expected_status" || true

  # Always capture pods JSON and describe text
  kubectl get pods -n "$ns" -o json > "${MOCK_DIR}/${prefix}-get-pods.json"
  kubectl describe pod -n "$ns" --selector=app > "${MOCK_DIR}/${prefix}-describe.txt" || \
    kubectl describe pods -n "$ns" > "${MOCK_DIR}/${prefix}-describe.txt"

  # Optional extra commands per scenario
  for cmd in "${extra_commands[@]}"; do
    case "$cmd" in
      logs)
        local pod
        pod=$(kubectl get pods -n "$ns" -o jsonpath='{.items[0].metadata.name}')
        kubectl logs -n "$ns" "$pod" --previous --tail=100 > "${MOCK_DIR}/${prefix}-logs.txt" 2>&1 || \
          echo "(no previous logs available)" > "${MOCK_DIR}/${prefix}-logs.txt"
        ;;
      endpoints)
        kubectl get endpoints -n "$ns" -o json > "${MOCK_DIR}/${prefix}-get-endpoints.json"
        ;;
    esac
  done

  echo "Captured $prefix files in $MOCK_DIR"
}

# Scenario 1: ImagePullBackOff
capture_scenario 01-image-pull-backoff.yaml k8s-trouble-image-pull 01-image-pull "ImagePullBackOff\|ErrImagePull"

# Scenario 2: CrashLoopBackOff (note: prefix is 02-crashloop2 to avoid collision with existing crashloop mock)
capture_scenario 02-crashloop-backoff.yaml k8s-trouble-crashloop 02-crashloop2 "CrashLoopBackOff" logs

# Scenario 3: OOMKilled
capture_scenario 03-oom-killed.yaml k8s-trouble-oom 03-oom "OOMKilled\|CrashLoopBackOff"

# Scenario 4: Liveness probe failure
capture_scenario 04-liveness-probe.yaml k8s-trouble-liveness 04-liveness "CrashLoopBackOff"

# Scenario 5: Missing secret
capture_scenario 05-missing-secret.yaml k8s-trouble-secret 05-missing-secret "CreateContainerConfigError\|ContainerCreating"

# Scenario 6: Port mismatch (pod runs healthy; failure surfaces via endpoints)
capture_scenario 06-port-mismatch.yaml k8s-trouble-port 06-port-mismatch "Running" endpoints

echo ""
echo "All scenario captures complete. Mock files in: $MOCK_DIR"
echo ""
echo "To clean up the test scenarios:"
echo "  for ns in k8s-trouble-image-pull k8s-trouble-crashloop k8s-trouble-oom k8s-trouble-liveness k8s-trouble-secret k8s-trouble-port; do"
echo "    kubectl delete namespace \$ns"
echo "  done"
```
