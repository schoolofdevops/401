    **nginx.conf** (`reference-app/dashboard/nginx.conf`):
    ```nginx
    server {
        listen 3000;
        root /usr/share/nginx/html;
        index index.html;

        # SPA fallback
        location / {
            try_files $uri $uri/ /index.html;
        }

        # Proxy API calls to api-gateway service (K8s service name)
        location /api/ {
            proxy_pass http://api-gateway:8080/;
            proxy_set_header Host $host;
        }

        # Direct service health/version polling
        location ~ ^/(api-gateway|catalog|worker)/ {
            # Rewrite to remove service name prefix, proxy to correct service
            # In K8s, use service names as hostnames
            rewrite ^/api-gateway/(.*) /$1 break;
            proxy_pass http://api-gateway:8080;
        }
        location ~ ^/catalog/ {
            rewrite ^/catalog/(.*) /$1 break;
            proxy_pass http://catalog:8081;
        }
        location ~ ^/worker/ {
            rewrite ^/worker/(.*) /$1 break;
            proxy_pass http://worker:8082;
        }
    }
    ```
